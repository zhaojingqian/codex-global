Java 客户端连接 MCP Server 代码参考

代码示例

package com.alibaba.mcp.test

import com.fasterxml.jackson.databind.ObjectMapper
import io.modelcontextprotocol.client.McpClient
import io.modelcontextprotocol.client.transport.WebClientStreamableHttpTransport
import io.modelcontextprotocol.client.transport.WebFluxSseClientTransport
import io.modelcontextprotocol.json.jackson.JacksonMcpJsonMapper
import io.modelcontextprotocol.spec.McpSchema
import kotlinx.coroutines.reactor.awaitSingleOrNull
import org.springframework.web.reactive.function.client.WebClient
import java.time.Duration

// 同步技术栈请使用
import io.modelcontextprotocol.client.transport.HttpClientSseClientTransport
import io.modelcontextprotocol.client.transport.HttpClientStreamableHttpTransport


/**
 * MCP Java SDK 连接 Aone 开放平台上的 MCP Server 代码片段
 *
 * 重点：
 * - baseUrl 填写 https://mcp.alibaba-inc.com
 * - sseEndpoint 填写 `"/{server}/sse"`
 *     * 如果是 Streamable HTTP Transport，则 endpoint 参数填写 `"/{server}/mcp"`
 * - 通过 WebClient 来配置请求头，包括标准的 Authorization 请求头和自定义的请求头
 * - 下面的代码使用 webflux 异步技术栈作为代码示例，webmvc 同步技术栈下请替换为相应的类
 *     * 例如 `HttpClientStreamableHttpTransport`、`HttpClientSseClientTransport`
 *     * 同时 HTTP 客户端也需要换成 java.net.http.HttpClient
 * - 重要：client 用完后，一定要进行 close !
 */
class AoneMcpDemo {
    suspend fun demo() {
        val objectMapper = ObjectMapper() // 这里最好换成 Spring 应用中自动装配的 ObjectMapper
        val mcpJsonMapper = JacksonMcpJsonMapper(objectMapper)

        val webClientBuilder = WebClient.builder()
            .baseUrl("https://mcp.alibaba-inc.com")
            .defaultHeaders { headers ->
                // 设置 OAuth2 授权生成的 access_token
                headers.setBearerAuth("mcpa_xxx")

                // 设置自定义的 header（可选）
                headers.set("X-Custome-Header", "xxx")
            }

        val sseTransport = WebFluxSseClientTransport(
            webClientBuilder,
            mcpJsonMapper,
            "/code/sse", // code 表示代码平台 MCP Server，也可以换成其他的 server 名称
        )

        // Streamable HTTP Transport 和 SSE Transport 二选一
        // 推荐优先使用 Streamable HTTP Transport
        val streamableHttpTransport = WebClientStreamableHttpTransport
            .builder(webClientBuilder)
            .jsonMapper(mcpJsonMapper)
            .endpoint("/code/mcp") // code 表示代码平台 MCP Server，也可以换成其他的 server 名称
            .build()

        val client = McpClient.async(streamableHttpTransport)
            .initializationTimeout(Duration.ofSeconds(60))
            .requestTimeout(Duration.ofSeconds(60))
            .build()

        client.initialize().awaitSingleOrNull()

        val tools = client.listTools().awaitSingleOrNull()
        println("MCP Server 中包含以下工具: $tools")

        val request = McpSchema.CallToolRequest("tool1", mapOf("arg1" to "foo", "arg2" to "bar"))
        val result = client.callTool(request).awaitSingleOrNull()
        println("tool1 工具调用结果: $result")

        // 重要：client 用完后，一定要进行 close !
        client.close()
    }
}
